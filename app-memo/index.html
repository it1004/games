구글 스프레드시트 연동 설정

구글 앱스 스크립트 배포 시 얻은 `https://script.google.com/macros/s/.../exec` 주소를 입력해 주세요.

📋 Google Apps Script (GAS) 코드

/** @OnlyCurrentDoc */

// 🔑 [1단계 필수] 권한 승인용 테스트 함수
// 앱스 스크립트 상단에서 testSetup을 선택하고 [▶ 실행]을 눌러 권한 승인을 완료하세요!
function testSetup() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["id", "title", "content", "category", "color", "isPinned", "createdAt", "updatedAt"]);
  }
  Logger.log("스프레드시트 연결 성공!");
}

function doGet(e) {
  var action = e ? e.parameter.action : null;
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["id", "title", "content", "category", "color", "isPinned", "createdAt", "updatedAt"]);
  }
  
  if (action === "read") {
    var data = sheet.getDataRange().getValues();
    var rows = data.slice(1);
    var memos = rows.map(function(row) {
      return {
        id: String(row[0]),
        title: String(row[1]),
        content: String(row[2]),
        category: String(row[3]),
        color: String(row[4]),
        isPinned: row[5] === true || String(row[5]).toLowerCase() === "true",
        createdAt: String(row[6]),
        updatedAt: String(row[7])
      };
    });
    return ContentService.createTextOutput(JSON.stringify({ status: "success", data: memos }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "API Ready" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["id", "title", "content", "category", "color", "isPinned", "createdAt", "updatedAt"]);
    }
    
    if (action === "create") {
      var memo = contents.memo;
      sheet.appendRow([memo.id, memo.title, memo.content, memo.category, memo.color, memo.isPinned, memo.createdAt, memo.updatedAt]);
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "update") {
      var memo = contents.memo;
      var data = sheet.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(memo.id)) {
          sheet.getRange(i + 1, 1, 1, 8).setValues([[memo.id, memo.title, memo.content, memo.category, memo.color, memo.isPinned, memo.createdAt, memo.updatedAt]]);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "delete") {
      var id = contents.id;
      var data = sheet.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(id)) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

🚀 승인 차단 해결 및 연동 순서

  1. 접속 위치 확인: 반드시 구글 시트 상단 [확장 프로그램] ➔ [Apps Script]로 접속해야 합니다. (별도 탭으로 연 `script.google.com`에서 만드시면 차단됩니다)
  2. 위의 코드를 맨 위의 /** @OnlyCurrentDoc */ 주석까지 **전체 복사**하여 붙여넣고 저장(Ctrl+S)합니다.
  3. 🔑 권한 승인: 상단 함수 드롭다운에서 testSetup을 선택하고 ▶ 실행 버튼을 누릅니다.
  4. 승인 창 우회: [권한 검토] ➔ [계정 선택] ➔ 좌측 하단 고급 (Advanced) ➔ [프로젝트명](으)로 이동(안전하지 않음) ➔ [허용 (Allow)] 클릭하면 차단 없이 승인됩니다.
  5. 우측 상단 [배포] ➔ [새 배포] ➔ 톱니바퀴 ➔ '웹 앱 (Web App)' 선택 후, 액세스 권한: '모든 사용자 (Anyone)' 설정하여 배포 URL을 생성합니다.

🚨 계속해서 권한 오류나 차단이 발생하는 경우 Check-List

  • 다중 구글 계정 로그인 충돌: 크롬 브라우저에 구글 계정이 여러 개 로그인되어 있으면 앱스 스크립트 권한 승인이 꼬입니다. 크롬 시크릿 창(Ctrl+Shift+N)을 열어서 해당 구글 계정 1개만 로그인 후 진행해 보세요.
  • 회사/학교(Workspace) 계정 정책: 조직 보안 정책상 외부 연결이 차단된 경우일 수 있으니, **개인 구글 계정(@gmail.com)**으로 진행해 주시기 바랍니다.
